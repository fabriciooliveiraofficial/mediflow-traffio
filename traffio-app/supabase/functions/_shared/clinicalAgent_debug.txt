/**
 * Normalizes a Brazilian WhatsApp phone number to E.164 format: 55 + DDD (2) + number (8 or 9) = 12–13 digits.
 * Accepts any format: "(41) 99775-9569", "41997759569", "5541997759569", etc.
 * Returns null if the number is clearly invalid (wrong digit count).
 */
function normalizePhone(raw: string): { phone: string } | { error: string } {
  const digits = raw.replace(/\D/g, "");
  if (digits.length < 7) {
    return { error: `Número muito curto (${digits.length} dígitos). Exemplo: 41 99775-9569 ou +1 404 925-7024.` };
  }

  // ── Already has known international prefix → keep as-is ──────────────────
  // USA/Canada (+1): 1 + 10 digits = 11
  if (digits.startsWith("1") && digits.length === 11) return { phone: digits };
  // Argentina (+54): 54 + 10-12 digits
  if (digits.startsWith("54") && digits.length >= 12 && digits.length <= 14) return { phone: digits };
  // Portugal (+351): 351 + 9 digits = 12
  if (digits.startsWith("351") && digits.length === 12) return { phone: digits };
  // UK (+44): 44 + 10 digits = 12
  if (digits.startsWith("44") && digits.length >= 11 && digits.length <= 13) return { phone: digits };
  // Spain (+34): 34 + 9 digits = 11
  if (digits.startsWith("34") && digits.length === 11) return { phone: digits };
  // France (+33): 33 + 9 digits = 11
  if (digits.startsWith("33") && digits.length === 11) return { phone: digits };
  // Italy (+39): 39 + 9-10 digits
  if (digits.startsWith("39") && digits.length >= 11 && digits.length <= 12) return { phone: digits };
  // Germany (+49): 49 + 10-11 digits
  if (digits.startsWith("49") && digits.length >= 11 && digits.length <= 13) return { phone: digits };
  // Mexico (+52): 52 + 10 digits = 12
  if (digits.startsWith("52") && digits.length >= 12 && digits.length <= 14) return { phone: digits };
  // Colombia (+57): 57 + 10 digits = 12
  if (digits.startsWith("57") && digits.length === 12) return { phone: digits };
  // Chile (+56): 56 + 9 digits = 11
  if (digits.startsWith("56") && digits.length >= 11 && digits.length <= 12) return { phone: digits };

  // ── Brazil (+55) ─────────────────────────────────────────────────────────
  // Strip country code 55 if present to normalize
  let base = digits;
  if (base.startsWith("55") && (base.length === 12 || base.length === 13)) {
    base = base.slice(2);
  }
  // Brazilian local: DDD (2 digits) + number (8 landline or 9 mobile) = 10 or 11
  if (base.length === 10 || base.length === 11) {
    return { phone: "55" + base };
  }

  // ── Generic international: if 11+ digits and starts with plausible country code, accept ──
  if (digits.length >= 11 && digits.length <= 15) {
    return { phone: digits };
  }

  return { error: `Número inválido (${digits.length} dígitos). Verifique o código do país e DDD. Exemplos: 41 99775-9569 (BR), +1 404 925-7024 (US).` };
}

/**
 * Formats a normalized E.164-like digit string into a pretty display version.
 * Handles USA (+1) and Brazil (+55) specifically, with a generic fallback.
 */
function formatPhone(digits: string | null | undefined): string {
  if (!digits) return "N/A";
  const raw = digits.replace(/\D/g, "");

  // USA/Canada (+1): 1 + 10 digits = 11
  if (raw.startsWith("1") && raw.length === 11) {
    return `+1 (${raw.slice(1, 4)}) ${raw.slice(4, 7)}-${raw.slice(7, 11)}`;
  }

  // Brazil (+55): 55 + 10 or 11 digits
  if (raw.startsWith("55")) {
    const ddd = raw.slice(2, 4);
    const rest = raw.slice(4);
    if (rest.length === 9) { // Mobile: 9XXXX-XXXX
      return `+55 (${ddd}) ${rest.slice(0, 5)}-${rest.slice(5, 9)}`;
    }
    if (rest.length === 8) { // Landline: XXXX-XXXX
      return `+55 (${ddd}) ${rest.slice(0, 4)}-${rest.slice(4, 8)}`;
    }
    return `+55 (${ddd}) ${rest}`;
  }

  // Generic international fallback: +XX XXXXX-XXXX
  if (raw.length >= 12) {
    return `+${raw.slice(0, 2)} ${raw.slice(2, 7)}-${raw.slice(7)}`;
  }

  return raw; // Fallback to raw if no match
}

/**
 * Masks a CPF for safe display: 96528168172 → 965.***.***-72
 */
function maskCpf(cpf: string | null | undefined): string {
  if (!cpf) return "***.***.***-**";
  const digits = cpf.replace(/\D/g, "");
  if (digits.length < 11) return "***.***.***-**";
  return `${digits.slice(0, 3)}.***.***-${digits.slice(9, 11)}`;
}

/**
 * Masks an email for safe display: fabricio@traffio.com.br → fabr*******@traffio.com.br
 */
function maskEmail(email: string | null | undefined): string {
  if (!email) return "***@***.***";
  const [local, domain] = email.split("@");
  if (!domain) return "***@***.***";
  const visible = local.slice(0, Math.min(4, local.length));
  return `${visible}${'*'.repeat(Math.max(1, local.length - visible.length))}@${domain}`;
}

/** Simple string hash for loop detection (FNV-1a) */
function quickHash(str: string): string {
  let hash = 2166136261;
  for (let i = 0; i < str.length; i++) {
    hash ^= str.charCodeAt(i);
    hash = (hash * 16777619) >>> 0;
  }
  return hash.toString(36);
}

export class ClinicalAgent {
  private supabase: SupabaseClient;
  private config: AgentConfig;
  private today: string;
  public humanHandoffRequested = false;
  public bookingCompleted = false; // Plan B: track if a conversion happened in this turn
  /** Tracks whether an intermediate message was already sent in this turn */
  private _intermediateMessageSent = false;
  /** Tracks phase-appropriate tools for this turn */
  private _phaseTools: typeof TOOLS[number][] = [...TOOLS];
  /** Current specialist handling the turn */
  private _currentSpecialist: SpecialistType = 'ROUTER';
  /** Mutable session context — updated during the turn, persisted by caller */
  public sessionContext: Record<string, any>;

  constructor(supabase: SupabaseClient, config: AgentConfig) {
    this.supabase = supabase;
    this.config = config;
    this.sessionContext = { ...(config.sessionContext ?? {}) };

    // Compute today in Brazil timezone (Edge Function runs in UTC)
    const now = new Date();
    const brazilParts = new Intl.DateTimeFormat("en-CA", {
      timeZone: "America/Sao_Paulo",
      year: "numeric", month: "2-digit", day: "2-digit"
    }).format(now).split("-").map(Number);
    const [y, m, d] = brazilParts;
    this.today = `${y}-${String(m).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
  }

  private getTemporalContext(): string {
    const now = new Date();
    const formatter = new Intl.DateTimeFormat("pt-BR", {
      timeZone: "America/Sao_Paulo",
      weekday: "long",
      year: "numeric",
      month: "long",
      day: "numeric",
      hour: "2-digit",
      minute: "2-digit",
      hour12: false
    });
    
    return `
# CONTEXTO TEMPORAL (CRÍTICO)
- Data atual (Brasília): ${formatter.format(now)}
- Data para ferramentas (YYYY-MM-DD): ${this.today}
- Regra: SEMPRE ofereça as vagas mais próximas (ex: hoje, amanhã) antes de sugerir datas distantes.
`;
  }

  /**
   * Specialist types for Multi-Agent Orchestration
   */
  private getSpecialistPrompt(type: SpecialistType): string {
    switch (type) {
      case 'TRIAGE':
        return `
# ESPECIALISTA EM TRIAGEM E IDENTIFICAÇÃO
Seu foco é acolher o paciente, descobrir o nome dele e coletar dados básicos (CPF, E-mail, Nascimento) se necessário.
- **OBJETIVO**: Capturar o lead (save_lead_info) e identificar o paciente (lookup_patient).
- **TOM**: Extremamente acolhedor, empático e focado em conexão pessoal.
- **AÇÃO**: NUNCA agende sem o nome. Colete dados UM POR VEZ.
`;
      case 'BOOKING':
        return `
# ESPECIALISTA EM AGENDAMENTO (CONVERSOR DE ELITE)
Seu foco é converter o interesse do paciente em um horário confirmado.
- **OBJETIVO**: Encontrar médicos, serviços e horários (list_doctors, get_next_available_date, book_appointment).

  /**
   * Intent Orchestrator: Classified the user message into a specialist node.
   * Uses a fast LLM call (gpt-4o-mini) for low latency.
   */
  private async routeIntent(userMessage: string): Promise<SpecialistType> {
    const history = this.config.conversationHistory.slice(-4).map(m => `${m.role}: ${m.content}`).join("\n");
    const prompt = `
Determine which specialist should handle the next turn.
USER MESSAGE: "${userMessage}"
HISTORY:
${history}

SPECIALISTS:
- TRIAGE: Introductions, getting patient name, collecting CPF/email/birthdate.
- BOOKING: Looking for doctors, specialties, dates, times, or confirming a selection.
- RAG_FAQ: Questions about exams, procedures, price (general), insurance accepted, clinic location info, parking, wifi.
- SALES: Handling objections like "it's expensive", "I'll think about it", "don't have my time".
- SUPPORT: Cancellations, rescheduling, checking existing appointments.

RESPONSE (JSON ONLY):
{"specialist": "TRIAGE" | "BOOKING" | "RAG_FAQ" | "SALES" | "SUPPORT"}
`;

    try {
      const resp = await fetch("https://api.openai.com/v1/chat/completions", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${this.config.apiKey}`
        },
        body: JSON.stringify({
          model: "gpt-4o-mini", // Use mini for router
          messages: [{ role: "system", content: "You are a specialist router for a dental clinic." }, { role: "user", content: prompt }],
          response_format: { type: "json_object" },
          temperature: 0
        })
      });
      const data = await resp.json();
      const choice = JSON.parse(data.choices[0].message.content);
      return choice.specialist as SpecialistType;
    } catch (e) {
      console.error("[ClinicalAgent] Router fail:", e);
      return 'BOOKING'; // Fallback
    }
  }

# ETAPA 2B: MICRO-COMPROMISSO PROGRESSIVO (COMMITMENT LADDER)

## PRINCÍPIO PSICOLÓGICO
Cada "sim" pequeno torna o próximo "sim" mais fácil. NUNCA peça o "grande sim" (confirmar agendamento) sem antes ter construído uma escada de micro-compromissos.

## A ESCADA DOS 5 SINS
1. **SIM de curiosidade** (custo zero): "Posso verificar os horários disponíveis para você?" → paciente diz sim
2. **SIM de preferência**: "Encontrei vagas! Prefere de manhã ou à tarde?" → paciente escolhe
3. **SIM de seleção**: "O das 10h na quinta fica bom?" → paciente confirma
4. **SIM de dados** (se necessário): "Só preciso do seu CPF para finalizar, pode ser?" → paciente fornece
5. **SIM de confirmação** (fechamento): "Posso confirmar seu agendamento?" → AGENDADO!

## TÉCNICA: ESCOLHA ENTRE DOIS SINS
NUNCA faça perguntas de sim/não. SEMPRE ofereça 2 opções positivas:
- ❌ "Quer agendar?" (sim/não — dá margem para "não")
- ✅ "Prefere terça ou quinta?" (sim/sim — ambas as respostas avançam)
- ❌ "Tem preferência de horário?" (aberta — gera hesitação)
- ✅ "Temos às 9h e às 14h — qual fica melhor para você?" (fechada — gera decisão)

## TÉCNICA: ASSUMPTIVE CLOSE (FECHAMENTO PRESUNTIVO)
Quando o paciente demonstra intenção clara, NÃO peça permissão — aja como se já estivesse decidido:
- ❌ "Gostaria de agendar com a Dra. Fabiola?"
- ✅ "Vou verificar os melhores horários com a Dra. Fabiola para você! 🔍"
- ❌ "Quer que eu procure vagas?"
- ✅ "Já estou procurando as vagas mais próximas! 😊"

## REGRA: NUNCA PERGUNTE ALGO QUE VOCÊ PODE RESOLVER
Se o paciente já disse o que quer → NÃO peça confirmação → VÁ BUSCAR
Se há apenas 1 opção → NÃO apresente como escolha → APRESENTE como solução
Se o contexto é óbvio → NÃO peça informação → ASSUMA e confirme

## SPIN SELLING ADAPTADO PARA SAÚDE (TÉCNICA BRASILEIRA DE ALTA CONVERSÃO)
Quando o paciente demonstra interesse mas ainda não decidiu agendar, use a sequência SPIN adaptada:

### S — SITUAÇÃO (entenda o contexto sem interrogar)
NÃO faça perguntas genéricas ("O que você precisa?"). Extraia a situação do que o paciente JÁ disse:
- Paciente disse "tô com dor no dente" → Situação: dor dental aguda, precisa de atendimento rápido
- Paciente disse "quero fazer uma avaliação" → Situação: interesse preventivo, pode ser convertido

### P — PROBLEMA (identifique a dor/necessidade real)
Faça UMA pergunta que revela a dor real (não a dor clínica, a dor EMOCIONAL):
- "Faz tempo que está sentindo isso?" (revela urgência)
- "Isso tem te incomodado no dia a dia?" (revela impacto emocional)
- "Já tentou resolver antes?" (revela frustração e urgência)
REGRA: Máximo 1 pergunta SPIN por turno. Não interrogue o paciente.

### I — IMPLICAÇÃO (mostre o custo de não agir — SEM alarmismo)
Use APENAS quando o paciente hesita. NUNCA use na primeira interação:
- "Entendo! Muitos pacientes que esperaram acabaram precisando de um tratamento mais extenso depois."
- "Quanto antes a gente avalia, mais opções de tratamento temos — e geralmente fica mais simples."

### N — NECESSIDADE DE SOLUÇÃO (conduza ao agendamento como a solução natural)
Apresente o agendamento como a SOLUÇÃO para o problema, não como uma venda:
- "O primeiro passo é fazer uma avaliação — em [X] minutos o(a) Dr(a). vai ter um panorama completo e te orientar. Posso verificar os horários?"
- "A melhor forma de resolver isso é com uma consulta. Já tenho horários disponíveis essa semana!"

### REGRA SPIN: Use as etapas NATURALMENTE na conversa — NUNCA de forma mecânica ou sequencial forçada. O paciente não pode perceber que existe uma técnica sendo aplicada.

  /** Returns UUIDs of active doctors for this tenant via the doctors table. */
  private async getTenantDoctorIds(): Promise<string[]> {
    const { data } = await this.supabase
      .from("doctors")
      .select("id")
      .eq("tenant_id", this.config.tenantId)
      .eq("is_active", true);
    return (data || []).map((d: any) => d.id).filter(Boolean);
  }

  /**
   * Resolves a doctor's name or ID to a valid UUID.
   * If ID is already a UUID, returns it. If not, searches by full_name.
   */
  private async resolveDoctorId(idOrName: string | null | undefined): Promise<string | { error: string }> {
    if (!idOrName) return { error: "ID ou nome do médico não informado." };
    if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(idOrName)) {
      return idOrName;
    }
    const tenantDocIds = await this.getTenantDoctorIds();

    // Query profiles directly (root) and join members + doctors
    const { data: byName } = await this.supabase
      .from("profiles")
      .select("id")
      .ilike("full_name", `%${idOrName}%`)
      .in("id", tenantDocIds)
      .limit(1);

    if (!byName?.[0]) return { error: `Médico "${idOrName}" não encontrado. Use list_doctors primeiro.` };
    return byName[0].id;
  }

  private async executeTool(name: string, args: any): Promise<string> {
    try {
      switch (name) {
        case "lookup_patient": {
          let phone = this.config.patientPhone;
          if (args.phone) {
            const norm = normalizePhone(args.phone);
            if ("error" in norm) return JSON.stringify({ found: false, error: norm.error });
            phone = norm.phone;
          }
          const { data } = await this.supabase
            .from("patients")
            .select("id, full_name, birth_date, cpf, email, phone")
            .eq("tenant_id", this.config.tenantId)
            .eq("phone", phone)
            .maybeSingle();
          if (data) {
            // Persist first name in session so it survives across turns
            if (data.full_name) this.sessionContext.known_first_name = data.full_name.split(" ")[0];
            return JSON.stringify({
              found: true,
              patient_id: data.id,
              name: data.full_name,
              birth_date: data.birth_date,
              cpf: data.cpf,
              email: data.email,
              cpf_masked: maskCpf(data.cpf),
              email_masked: maskEmail(data.email),
              phone_formatted: formatPhone(data.phone)
            });
          }
          return JSON.stringify({ found: false, phone, phone_formatted: formatPhone(phone) });
        }

        case "update_patient_details": {
          const updateData: any = {};
          if (args.name) updateData.full_name = args.name;
          if (args.cpf) {
            const cleanCpf = String(args.cpf).replace(/\D/g, "");
            if (cleanCpf.length === 11) {
              updateData.cpf = cleanCpf;
              this.sessionContext.collected_cpf = cleanCpf;
            } else {
              return JSON.stringify({ success: false, error: "CPF deve conter exatamente 11 dígitos numéricos." });
            }
          }
          if (args.email) {
            updateData.email = args.email;
            this.sessionContext.collected_email = args.email;
          }
          if (args.birth_date) {
            const resolvedBirth = resolveDate(args.birth_date) ?? args.birth_date;
            updateData.birth_date = resolvedBirth;
            this.sessionContext.collected_birth_date = resolvedBirth;
          }

          if (Object.keys(updateData).length === 0) return JSON.stringify({ success: false, error: "Nenhum dado informado para atualização." });

          // Resolve patient ID: use arg if provided, otherwise resolve from session phone
          let targetPatientId = args.patient_id;
          if (!targetPatientId) {
            const { data: p } = await this.supabase
              .from("patients").select("id")
              .eq("tenant_id", this.config.tenantId)
              .eq("phone", this.config.patientPhone)
              .maybeSingle();
            targetPatientId = p?.id;
          }

          if (!targetPatientId) return JSON.stringify({ success: false, error: "Paciente não identificado para atualização." });

          const { error } = await this.supabase
            .from("patients")
            .update(updateData)
            .eq("id", targetPatientId)
            .eq("tenant_id", this.config.tenantId);

          if (error) return JSON.stringify({ success: false, error: error.message });
          return JSON.stringify({ success: true });
        }

        case "register_patient": {
          // Always prefer the session phone (from Z-API) — it's the canonical format.
          // Only use args.phone if explicitly booking for a DIFFERENT person.
          let phone = this.config.patientPhone;
          if (args.phone) {
            const norm = normalizePhone(args.phone);
            if ("error" in norm) return JSON.stringify({ success: false, error: norm.error });
            phone = norm.phone;
          }

          // Dedup guard: check if patient with this phone already exists
          const { data: existing } = await this.supabase
            .from("patients")
            .select("id, full_name")
            .eq("tenant_id", this.config.tenantId)
            .eq("phone", phone)
            .maybeSingle();

          // Persist collected data in session for later multi-step use
          if (args.cpf) {
            const cleanCpf = String(args.cpf).replace(/\D/g, "");
            if (cleanCpf.length === 11) {
              this.sessionContext.collected_cpf = cleanCpf;
            }
          }
          if (args.email) this.sessionContext.collected_email = args.email;
          if (args.birth_date) {
            const resolvedBirth = resolveDate(args.birth_date) ?? args.birth_date;
            this.sessionContext.collected_birth_date = resolvedBirth;
          }

          // Check if patient with this exact CPF already exists for this tenant
          if (this.sessionContext.collected_cpf) {
            const { data: cpfExisting } = await this.supabase
              .from("patients")
              .select("id, full_name")
              .eq("tenant_id", this.config.tenantId)
              .eq("cpf", this.sessionContext.collected_cpf)
              .maybeSingle();
            
            if (cpfExisting) return JSON.stringify({ success: false, error: `Já existe um cadastro com o CPF ${this.sessionContext.collected_cpf} para o paciente ${cpfExisting.full_name}. Verifique os dados.` });
          }

          if (existing) {
            // Patient already exists — update with ALL provided fields
            const updatePayload: Record<string, any> = {};
            if (args.name && args.name !== existing.full_name) updatePayload.full_name = args.name;
            if (args.cpf) updatePayload.cpf = args.cpf;
            if (args.email) updatePayload.email = args.email;
            if (args.birth_date) updatePayload.birth_date = args.birth_date;

            if (Object.keys(updatePayload).length > 0) {
              await this.supabase.from("patients").update(updatePayload).eq("id", existing.id);
            }

            if (args.name) this.sessionContext.known_first_name = args.name.split(" ")[0];
            return JSON.stringify({ success: true, patient_id: existing.id, name: args.name || existing.full_name, note: "Paciente já cadastrado — dados atualizados." });
          }

          const { data, error } = await this.supabase
            .from("patients")

              tenant_id: this.config.tenantId,

              cpf: args.cpf || null,
              email: args.email || null,
              birth_date: args.birth_date || null
            })
            .select("id")
            .single();
          if (error) return JSON.stringify({ success: false, error: error.message });
          // Persist first name in session
          if (args.name) this.sessionContext.known_first_name = args.name.split(" ")[0];
          return JSON.stringify({ success: true, patient_id: data.id, name: args.name });

        case "get_next_available_date": {

          // V2: location-aware query (180-day window)

            timeZone: "America/Sao_Paulo",

          const bufferMinutes = curHH * 60 + curMM + BOOKING_BUFFER_MINUTES;

            p_doctor_id: gnDoctorId,

            p_limit: 5,
            p_duration_minutes: 30,
            p_current_time: brazilTimeStr // Passa a hora atual de Brasília para o SQL
          });

          if (error) {
            console.error("[ClinicalAgent] find_next_available_dates error:", error.message);
            return JSON.stringify({ dates: [], note: "Não foi possível verificar disponibilidade." });
          }

          const dates = Array.isArray(data) ? data : [];

          if (dates.length === 0) {
            return JSON.stringify({
              dates: [],
              note: `Não há disponibilidade nos próximos 180 dias para este médico a partir de ${fromDateStr}. Gostaria de entrar na Lista de Espera?`,
              max_date: maxDateStr
            });
          }

          const filterByBuffer = (slots: string[]) => {
            if (!isToday) return slots;
            return (slots ?? []).filter((t: string) => {
              const [hh, mm] = t.split(':').map(Number);
              return hh * 60 + mm >= bufferMinutes;
            });
          };

          // Period Filter
          const filterByPeriod = (slots: string[], period: string) => {
            if (!period || period === "any") return slots;
            return slots.filter((t: string) => {
              const hour = parseInt(t.split(':')[0]);
              if (period === "morning") return hour < 12;
              if (period === "afternoon") return hour >= 12 && hour < 18;
              if (period === "evening") return hour >= 18;
              return true;
            });
          };

          const filteredDates = dates.map((match: any) => ({
            ...match,
            prime_slots:   filterByPeriod(filterByBuffer(match.prime_slots   ?? []), args.preferred_period),
            regular_slots: filterByPeriod(filterByBuffer(match.regular_slots ?? []), args.preferred_period),

          const [curHH, curMM] = brazilTimeStr.split(':').map(Number);

          const bufferMinutes = curHH * 60 + curMM + BOOKING_BUFFER_MINUTES;

            let validSlots = allSlots;
            
            // Apply 2h buffer for today

               validSlots = validSlots.filter((timeStr: string) => {
                  const [hh, mm] = timeStr.split(':').map(Number);
                  return hh * 60 + mm >= bufferMinutes;
               });
            }
            return { ...match, slots: validSlots };
          }).filter(match => match.slots.length > 0);

          // Anti-hallucination guard: If earliest date is > 60 days, warn

              });
            }

                 dates: exactDateMatches,
                 note: `Encontrados horários para a data solicitada (${resolvedDate}).`
             });
          }

            date: d.date,
            location_id: d.location_id,
            location_name: d.location_name,
            slots: d.slots ?? []
          }));
          this.sessionContext.last_offered_doctor_id = caDoctorId;

          return JSON.stringify({
            dates: filteredDates.slice(0, 3),
            note: `Sem horários exatos em ${resolvedDate}. Essas são as próximas datas e unidades disponíveis.`

          if (args.patient_phone) {
            const norm = normalizePhone(args.patient_phone);
            if ("error" in norm) return JSON.stringify({ success: false, error: norm.error });
            bookingPhone = norm.phone;
          }
          let patientId: string | null = null;
          const { data: existingPatient } = await this.supabase
            .from("patients").select("id")
            .eq("tenant_id", this.config.tenantId)
            .eq("phone", bookingPhone)
            .maybeSingle();

          if (existingPatient) {
            patientId = existingPatient.id;
          } else {
            if (!args.patient_name) return JSON.stringify({ success: false, error: "Nome do paciente não informado." });
            const { data: newPatient, error: insertErr } = await this.supabase
              .from("patients")
              .insert({
                tenant_id: this.config.tenantId,
                phone: bookingPhone,
                full_name: args.patient_name,
                cpf: this.sessionContext.collected_cpf || null,
                email: this.sessionContext.collected_email || null,
                birth_date: this.sessionContext.collected_birth_date || null
              })
              .select("id")
              .single();
            if (insertErr || !newPatient) return JSON.stringify({ success: false, error: "Não foi possível registrar o paciente." });
            patientId = newPatient.id;
          }

          if (!patientId) return JSON.stringify({ success: false, error: "Paciente não encontrado." });

          // Pre-booking conflict check — guard against race conditions and
          // stale availability data shown to the patient
          const { count: conflictCount } = await this.supabase
            .from("appointments")
            .select("id", { count: "exact", head: true })
            .eq("doctor_id", baDoctorId)
            .eq("date", resolvedDate)
            .eq("start_time", args.time)
            .not("status", "in", '("canceled","noshow","cancelled","no_show")');

          if (conflictCount && conflictCount > 0) {
            return JSON.stringify({
              success: false,
              error: "SLOT_CONFLICT",
              message: "Este horário acabou de ser preenchido por outro paciente. Por favor, escolha outro horário.",
              action: "call get_next_available_date to suggest alternatives"
            });
          }

          const { data, error } = await this.supabase.rpc("book_appointment", {
            p_tenant_id: this.config.tenantId,
            p_patient_id: patientId,
            p_doctor_id: baDoctorId,
            p_location_id: locationId,
            p_type_id: typeId,
            p_date: resolvedDate,
            p_start_time: args.time,
            p_booked_by: "ai_agent"
          });

          if (error) {
            console.error("[ClinicalAgent] book_appointment RPC error:", error.message);
            return JSON.stringify({ success: false, error: error.message });
          }

          const result = data as any;
          if (!result?.success) {
            const reason = result?.reason ?? result?.error ?? "unknown";
            if (reason === "slot_taken" || reason === "SLOT_CONFLICT") {
              return JSON.stringify({
                success: false,
                error: "SLOT_CONFLICT",
                message: "Este horário acabou de ser preenchido por outro paciente. Chame get_next_available_date para sugerir novas datas disponíveis."
              });
            }
            return JSON.stringify({ success: false, error: reason });
          }

          // Reset booking state in sessionContext
          this.sessionContext.current_doctor_id    = null;
          this.sessionContext.current_doctor_name  = null;
          this.sessionContext.current_service_id   = null;
          this.sessionContext.current_service_name = null;
          this.sessionContext.current_location_id  = null;
          this.sessionContext.current_location_name = null;
          this.sessionContext.last_offered_slots   = null;
          this.sessionContext.last_offered_doctor_id = null;

          this.bookingCompleted = true; // SINAL DE CONVERSÃO (Plan B)
          return JSON.stringify(data);
        }

        case "list_patient_appointments": {
          let phone = this.config.patientPhone;
          if (args.patient_phone) {
            const norm = normalizePhone(args.patient_phone);
            if ("error" in norm) return JSON.stringify({ appointments: [], error: norm.error });
            phone = norm.phone;
          }
          const { data: pat } = await this.supabase
            .from("patients").select("id")
            .eq("tenant_id", this.config.tenantId).eq("phone", phone).maybeSingle();
          if (!pat) return JSON.stringify({ appointments: [], note: "Paciente não encontrado." });

          const { data } = await this.supabase
            .from("appointments")
            .select("id, date, start_time, status, doctors(full_name), locations(name), appointment_types(name)")
            .eq("tenant_id", this.config.tenantId)
            .eq("patient_id", pat.id)
            .in("status", ["scheduled", "confirmed"])
            .order("date", { ascending: true })
            .order("start_time", { ascending: true })
            .limit(10);

          const appts = (data || []).map((a: any) => ({
            id: a.id,
            date: a.date,
            time: String(a.start_time).substring(0, 5),
            status: a.status,
            doctor: a.doctors?.full_name ?? "Médico",
            doctor_id: a.doctor_id,
            location: a.locations?.name ?? "Local",
            location_id: a.location_id,
            type: a.appointment_types?.name ?? "Consulta",
            type_id: a.type_id
          }));
          if (appts.length === 0) return JSON.stringify({ appointments: [], note: "Nenhum agendamento futuro encontrado." });
          return JSON.stringify({ appointments: appts });
        }

        case "cancel_appointment": {
          const { error } = await this.supabase
            .from("appointments")
            .update({ status: "canceled" })
            .eq("id", args.appointment_id)
            .eq("tenant_id", this.config.tenantId);
          if (error) return JSON.stringify({ success: false, error: error.message });
          return JSON.stringify({ success: true });
        }

        case "reschedule_appointment": {
          const newDate = resolveDate(args.new_date) ?? args.new_date;
          const { error } = await this.supabase
            .from("appointments")
            .update({ date: newDate, start_time: args.new_time, status: "scheduled" })
            .eq("id", args.appointment_id)
            .eq("tenant_id", this.config.tenantId);
          if (error) return JSON.stringify({ success: false, error: error.message });
          return JSON.stringify({ success: true, new_date: newDate, new_time: args.new_time });
        }

        case "verify_identity": {
          if (!this.config.patientCpf || !this.config.patientBirthDate) {
            return JSON.stringify({ verified: false, attempts_remaining: 0, note: "Cadastro base incompleto (CPF ou Data de Nascimento ausentes). O acesso não pode ser concedido sem esses dados no sistema." });
          }
          
          const dbCpf = this.config.patientCpf.replace(/\D/g, "");
          const dbYear = this.config.patientBirthDate.split("-")[0];
          
          const inCpf = (args.cpf_prefix || "").replace(/\D/g, "");
          const inYear = String(args.birth_year || "").trim();
          
          const verified = dbCpf.startsWith(inCpf) && dbYear === inYear;
          
          if (verified) {
             this.sessionContext.identity_verified    = true;
             this.sessionContext.identity_verified_at = new Date().toISOString();
             return JSON.stringify({ verified: true, note: "Identidade confirmada! Você pode prosseguir e fornecer as informações sensíveis solicitadas." });
          } else {
             return JSON.stringify({ verified: false, attempts_remaining: 2, note: "Dados incorretos. Peça para o paciente tentar novamente." });
          }
        }

        case "store_booking_selection": {
          if (!args.date || !args.time) return JSON.stringify({ error: "date e time são obrigatórios." });
          this.sessionContext.pending_booking = {
            date: args.date,
            time: args.time,
            location_id: args.location_id ?? this.sessionContext.current_location_id ?? null,
            doctor_id: this.sessionContext.current_doctor_id ?? null,
            type_id: this.sessionContext.current_service_id ?? null,
            patient_name: this.config.patientName ?? this.sessionContext.known_first_name ?? null
          };
          return JSON.stringify({ stored: true, note: "Seleção registrada. Agora mostre a mensagem de confirmação Etapa 4 ao paciente." });
        }

        case "save_lead_info": {
          const leadName = args.name?.trim();
          if (!leadName) return JSON.stringify({ success: false, error: "Nome não informado." });

          // Anti-hallucination check: name must be valid text, not just a number
          const nameWithoutDigits = leadName.replace(/\d/g, "").trim();
          if (nameWithoutDigits.length < 2) {
            console.warn(`[ClinicalAgent] Rejected invalid name: ${leadName}`);
            return JSON.stringify({ success: false, error: "O nome informado parece inválido. Por favor, poderia me dizer seu nome completo?" });
          }

          this.sessionContext.known_first_name = leadName.split(" ")[0];
          if (args.lead_temperature) this.sessionContext.lead_temperature = args.lead_temperature;
          if (args.interest) this.sessionContext.lead_interest = args.interest;
          if (args.phone_confirmed) this.sessionContext.phone_confirmed = true;

          const funnelData: any = {
            tenant_id: this.config.tenantId,
            patient_phone: this.config.patientPhone,
            patient_name: leadName,
            last_interaction_at: new Date().toISOString(),
          };
          if (args.lead_temperature) funnelData.lead_temperature = args.lead_temperature;
          if (args.interest) funnelData.lead_interest = args.interest;

          await this.supabase
            .from("patient_funnel_stage")
            .update(funnelData)
            .eq("tenant_id", this.config.tenantId)
            .eq("patient_phone", this.config.patientPhone);

          return JSON.stringify({
            success: true,
            note: "Lead salvo. Continue a conversa naturalmente — NÃO mencione ao paciente que salvou os dados."
          });
        }

        case "request_human_agent": {
          this.humanHandoffRequested = true;
          return JSON.stringify({ transferred: true });
        }

        default:
          return JSON.stringify({ error: `Ferramenta desconhecida: ${name}` });
      }
    } catch (e: any) {
      console.error(`[ClinicalAgent] Tool error (${name}):`, e.message);
      return JSON.stringify({ error: e.message, isTechnical: true });
    }
  }

  /**
   * Pre-router: detects simple confirmation/rejection intent BEFORE calling the LLM.
   */
  private preRoute(userMessage: string): string | { directBook: true } | null {
    const msg = userMessage.trim().toLowerCase();
    const patientName = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;

    const CONFIRM_YES = /^(sim|s|yes|confirmo|confirmar|confirma|pode|pode ser|tá|ta|ok|okay|certo|perfeito|ótimo|otimo|isso|exato|exatamente|pode confirmar|confirme)[\s!.,]*$/i;
    if (CONFIRM_YES.test(msg) && this.sessionContext.pending_booking?.date && this.sessionContext.pending_booking?.time) {
      return { directBook: true };
    }

    const SIMPLE_NO = /^(não|nao|nope|negativo|desistir|não quero|nao quero|não precisa|não vou|cancela|cancelar)[\s!.,]*$/i;
    if (SIMPLE_NO.test(msg) && this.sessionContext.pending_booking) {
      this.sessionContext.pending_booking = null;
      return `Sem problema, ${patientName || "tudo bem"}! 😊 Se quiser agendar depois ou precisar de outra coisa, é só me chamar. Como posso te ajudar?`;
    }

    const GREETING = /^(oi|olá|ola|bom dia|boa tarde|boa noite|hello|hi|e aí|eai|opa|hey)[\s!?.,]*$/i;
    if (GREETING.test(msg) && !this.sessionContext.current_doctor_id && !this.sessionContext.last_offered_slots) {
      if (!this.sessionContext.bot_introduced) return null;
      const h = new Date().getHours();
      const greeting = h < 12 ? "Bom dia" : h < 18 ? "Boa tarde" : "Boa noite";
      return `${greeting}${patientName ? `, ${patientName}` : ""}! 😊 Que bom falar com você! Como posso te ajudar hoje?`;
    }

    return null;
  }

  private getCurrentPhase(): 'GREET' | 'IDENTIFY' | 'SCHEDULE' | 'CONFIRM' | 'FAQ' | 'MANAGE' {
    const ctx = this.sessionContext;
    if (!ctx.bot_introduced || !ctx.name_captured) return 'GREET';
    if (ctx.reminder_cancel_requested || ctx.reminder_reschedule_requested) return 'MANAGE';
    if (ctx.pending_booking?.date && ctx.pending_booking?.time) return 'CONFIRM';
    if (ctx.current_doctor_id || ctx.current_service_id) return 'SCHEDULE';
    return 'SCHEDULE';
  }

  private getPhaseTools(): typeof TOOLS[number][] {
    const ctx = this.sessionContext;
    
    if (!ctx.bot_introduced || !ctx.name_captured) {
      return TOOLS.filter(t => 
        ['lookup_patient', 'save_lead_info'].includes(t.function.name)
      );
    }
    
    if (ctx.reminder_cancel_requested || ctx.reminder_reschedule_requested) {
      return TOOLS.filter(t => 
        ['list_patient_appointments', 'cancel_appointment', 'reschedule_appointment', 
         'verify_identity', 'check_availability', 'get_next_available_date'].includes(t.function.name)
      );
    }
    
    if (ctx.pending_booking?.date && ctx.pending_booking?.time) {
      return TOOLS.filter(t => 
        ['store_booking_selection', 'book_appointment', 'update_patient_details', 
         'register_patient', 'generate_pix_payment'].includes(t.function.name)
      );
    }
    
    if (ctx.current_doctor_id || ctx.current_service_id || ctx.last_offered_slots) {
      return TOOLS.filter(t => 
        ['list_appointment_types', 'list_service_locations', 'get_next_available_date',
         'check_availability', 'store_booking_selection', 'book_appointment',
         'list_doctor_insurances', 'add_to_waitlist', 'get_doctor_details',
         'register_patient', 'update_patient_details', 'save_lead_info',
         'request_human_agent', 'search_clinic_knowledge'].includes(t.function.name)
      );
    }
    
    return [...TOOLS];
  }

  private getSpecialistTools(type: SpecialistType): typeof TOOLS[number][] {
     switch (type) {
       case 'TRIAGE':
         return TOOLS.filter(t => ['lookup_patient', 'save_lead_info', 'register_patient', 'update_patient_details'].includes(t.function.name));
       case 'BOOKING':
         return TOOLS.filter(t => ['list_specialties', 'list_doctors', 'list_appointment_types', 'list_service_locations', 'get_next_available_date', 'check_availability', 'store_booking_selection', 'book_appointment', 'list_doctor_insurances', 'add_to_waitlist'].includes(t.function.name));
       case 'RAG_FAQ':
         return TOOLS.filter(t => ['search_clinic_knowledge', 'get_clinic_info', 'get_doctor_details', 'list_doctor_locations'].includes(t.function.name));
       case 'SALES':
         return TOOLS.filter(t => ['search_clinic_knowledge', 'list_doctor_insurances', 'get_doctor_details', 'generate_pix_payment'].includes(t.function.name));
       case 'SUPPORT':
         return TOOLS.filter(t => ['list_patient_appointments', 'cancel_appointment', 'reschedule_appointment', 'verify_identity', 'check_availability'].includes(t.function.name));
       default:
         return [...TOOLS];
     }
  }

  async processMessage(userMessage: string): Promise<string> {
    this._intermediateMessageSent = false;
    this.sessionContext.total_turns = (this.sessionContext.total_turns || 0) + 1;
    this.sessionContext.turns_since_last_tool_success = (this.sessionContext.turns_since_last_tool_success || 0) + 1;

    if (this.sessionContext.total_turns > 25 && !this.bookingCompleted) {
      this.humanHandoffRequested = true;
      const pName = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
      return `${pName ? `${pName}, ` : ""}percebi que nosso atendimento está demorando mais do que o normal. 😊 Vou chamar um de nossos atendentes para te ajudar de forma mais rápida. Um momento, por favor! 🙏`;
    }

    const preRouted = this.preRoute(userMessage);
    if (preRouted !== null) {
      if (typeof preRouted === "object" && preRouted.directBook) {
        const pb = this.sessionContext.pending_booking;
        const result = await this.executeTool("book_appointment", {
          patient_name: pb.patient_name ?? this.config.patientName,
          doctor_id: pb.doctor_id ?? this.sessionContext.current_doctor_id,
          location_id: pb.location_id ?? this.sessionContext.current_location_id,
          type_id: pb.type_id ?? this.sessionContext.current_service_id,
          date: pb.date,
          time: pb.time,
        });
        this.sessionContext.pending_booking = null;
        const parsed = JSON.parse(result);
        const pName = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
        if (parsed.success) {
          const doctorName = this.sessionContext.current_doctor_name ?? "médico";
          const locationName = this.sessionContext.current_location_name ?? "";
          return `✅ Agendamento confirmado${pName ? `, ${pName}` : ""}! 🎉\n\n👨‍⚕️ ${doctorName}\n📍 ${locationName}\n📅 ${pb.date.split("-").reverse().join("/")} às ⏰ ${pb.time}\n👤 ${pb.patient_name ?? this.config.patientName}\n\nA equipe aguarda você! Qualquer dúvida é só me chamar. 😊`;
        }
        if (parsed.error === "SLOT_CONFLICT") {
          return `Que pena${pName ? `, ${pName}` : ""}! 😕 Esse horário acabou de ser ocupado por outro paciente. Deixa eu verificar as próximas vagas disponíveis para você!`;
        }
      } else {
        return preRouted as string;
      }
    }

    this._currentSpecialist = await this.routeIntent(userMessage);
    const messages: any[] = [
      { role: "system", content: this.buildSystemPrompt(this._currentSpecialist) },
      ...this.config.conversationHistory.slice(-12).map(m => ({
        role: m.role === "assistant" ? "assistant" : "user",
        content: (m.content ?? "").substring(0, 500)
      })),
      { role: "user", content: userMessage }
    ];

    const bookingIntent = /\b(agendar|marcar|agendar|consulta|avaliação|avaliaç|implante|limpeza|tratamento|horário|horario|vaga|disponib)\b/i.test(userMessage);
    this._phaseTools = this.getSpecialistTools(this._currentSpecialist);

    for (let round = 0; round < 8; round++) {
      let data: any;
      try {
        const forceToolUse = (round === 0 && bookingIntent);
        data = await this.callOpenAI(messages, 0, forceToolUse);
      } catch (llmErr: any) {
        const pn = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
        return `${pn ? `${pn}, ` : ""}desculpe, tive um probleminha técnico! 😅 Pode repetir o que precisa? Estou aqui para te ajudar!`;
      }

      const choice = data.choices?.[0];
      if (!choice) {
        const pn = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
        return `${pn ? `${pn}, ` : ""}desculpe, tive um probleminha técnico! 😅 Pode repetir o que precisa?`;
      }

      const msg = choice.message;
      messages.push(msg);

      if (choice.finish_reason === "stop" || !msg.tool_calls?.length) {
        return msg.content ?? "Desculpe, não consegui processar sua solicitação.";
      }

      if (msg.content && msg.content.trim() && this.config.onIntermediateMessage) {
        if (!this._intermediateMessageSent) {
          await this.config.onIntermediateMessage(msg.content.trim()).catch(() => {});
          this._intermediateMessageSent = true;
        }
      }

      const STATEFUL_TOOLS = new Set([
        "list_specialties", "list_appointment_types", "list_service_locations",
        "get_next_available_date", "check_availability", "book_appointment",
        "list_doctor_insurances", "verify_identity", "register_patient", "update_patient_details",
        "save_lead_info"
      ]);
      const toolCalls = msg.tool_calls as ToolCall[];
      const hasStatefulCall = toolCalls.some(tc => STATEFUL_TOOLS.has(tc.function.name));

      let results: any[];
      if (hasStatefulCall || toolCalls.length === 1) {
        results = [];
        for (const tc of toolCalls) {
          const args = JSON.parse(tc.function.arguments);
          try {
            // Local tool timeout (5s) to avoid Edge Function timeout
            const toolResult = await Promise.race([
              this.executeTool(tc.function.name, args),
              new Promise<string>((_, reject) => setTimeout(() => reject(new Error("TOOL_TIMEOUT")), 5000))
            ]);
            results.push({ role: "tool", tool_call_id: tc.id, content: toolResult.substring(0, 3000) });
          } catch (toolErr: any) {
            console.error(`[ClinicalAgent] Tool ${tc.function.name} failed:`, toolErr.message);
            const errorMsg = toolErr.message === "TOOL_TIMEOUT" 
              ? JSON.stringify({ error: "TIMEOUT", isTechnical: true }) 
              : JSON.stringify({ error: "INTERNAL_ERROR", isTechnical: true });
            results.push({ role: "tool", tool_call_id: tc.id, content: errorMsg });
          }
        }
      } else {
        results = await Promise.all(
          toolCalls.map(async (tc) => {
            const args = JSON.parse(tc.function.arguments);
            try {
              const toolResult = await Promise.race([
                this.executeTool(tc.function.name, args),
                new Promise<string>((_, reject) => setTimeout(() => reject(new Error("TOOL_TIMEOUT")), 5000))
              ]);
              return { role: "tool", tool_call_id: tc.id, content: toolResult.substring(0, 3000) };
            } catch (toolErr: any) {
              console.error(`[ClinicalAgent] Parallel tool ${tc.function.name} failed:`, toolErr.message);
              const errorMsg = toolErr.message === "TOOL_TIMEOUT" 
                ? JSON.stringify({ error: "TIMEOUT", isTechnical: true }) 
                : JSON.stringify({ error: "INTERNAL_ERROR", isTechnical: true });
              return { role: "tool", tool_call_id: tc.id, content: errorMsg };
            }
          })
        );
      }

      // Track consecutive tool failures
      const failedTools = results.filter(r => {
        try { 
          const parsed = JSON.parse(r.content);
          return parsed.error && parsed.isTechnical === true; 
        } catch { return false; }
      });

      if (failedTools.length === toolCalls.length) {
        this.sessionContext.consecutive_tool_failures = (this.sessionContext.consecutive_tool_failures || 0) + 1;
        console.warn(`[ClinicalAgent] All tools failed this round. Consecutive failures: ${this.sessionContext.consecutive_tool_failures}`);
        
        if (this.sessionContext.consecutive_tool_failures >= 3) {
          console.warn(`[ClinicalAgent] 3+ consecutive tool failures — escalating to human`);
          this.humanHandoffRequested = true;
          const pName = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
          return `${pName ? `${pName}, ` : ""}desculpe, estou com uma dificuldade técnica no momento. 😊 Vou chamar um atendente para te ajudar. Um momento! 🙏`;
        }
      } else {
        this.sessionContext.consecutive_tool_failures = 0;
      }

      messages.push(...results);
    }

    // Round limit reached — graceful recovery instead of generic error.
    // Signal human handoff so the conversation is queued for a human agent.
    this.humanHandoffRequested = true;
    const pName = this.sessionContext.known_first_name ?? (this.config.patientName ?? "").split(" ")[0] ?? null;
    console.warn(`[ClinicalAgent] Round limit reached for ${this.config.patientPhone} — triggering human handoff`);
    return `${pName ? `${pName}, ` : ""}desculpe a demora! 😊 Sua solicitação é um pouquinho mais complexa e vou chamar um de nossos atendentes para te ajudar pessoalmente. Em instantes alguém entrará em contato. 🙏`;
  }

  private async callOpenAI(messages: any[], attempt = 0, forceToolUse = false): Promise<any> {
    console.log(`[ClinicalAgent] Calling OpenAI [${this.config.model}] Round ${attempt} — Messages: ${messages.length} — ForceTool: ${forceToolUse}`);
    
    const headers: Record<string, string> = {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${this.config.apiKey}`
    };

    // Inject LangSmith headers for observability
    const langsmithKey = Deno.env.get("LANGSMITH_API_KEY") || this.sessionContext.langsmith_api_key;
    if (langsmithKey) {
      headers["X-Langsmith-Api-Key"] = langsmithKey;
      headers["X-Langsmith-Project"] = Deno.env.get("LANGSMITH_PROJECT") || "traffio-clinical-agent";
    }

    const res = await fetch("https://api.openai.com/v1/chat/completions", {
      method: "POST",
      headers,
      body: JSON.stringify({
        model: this.config.model,
        messages,
        tools: this._phaseTools,
        tool_choice: forceToolUse ? "required" : "auto",
        parallel_tool_calls: true,
        temperature: 0.3
      })
    });

    if (res.status === 429 && attempt < 3) {
      // Respect Retry-After header if present, otherwise use exponential backoff
      const retryAfter = res.headers.get("retry-after");
      const waitMs = retryAfter ? parseFloat(retryAfter) * 1000 : (attempt + 1) * 2000;
      console.warn(`[ClinicalAgent] Rate limited (429). Retrying in ${waitMs}ms (attempt ${attempt + 1}/3)`);
      await new Promise((r) => setTimeout(r, waitMs));
      return this.callOpenAI(messages, attempt + 1, forceToolUse);
    }

    if (!res.ok) {
      const errText = await res.text();
      console.error(`[ClinicalAgent] OpenAI API error ${res.status}:`, errText);
      throw new Error(`OpenAI API error ${res.status}: ${errText}`);
    }

    return res.json();
  }
}
